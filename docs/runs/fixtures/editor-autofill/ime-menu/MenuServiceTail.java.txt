    private static int countExcludedTextNodes(AssistStructure tree) {
        int count = 0;
        for (int w = 0; w < tree.getWindowNodeCount(); w++) {
            count += countExcluded(tree.getWindowNodeAt(w).getRootViewNode(), false);
        }
        return count;
    }

    private static int countExcluded(ViewNode node, boolean ancestorExcluded) {
        if (node == null) return 0;
        int importance = node.getImportantForAutofill();
        boolean excludedHere = ancestorExcluded
            || importance == View.IMPORTANT_FOR_AUTOFILL_NO
            || importance == View.IMPORTANT_FOR_AUTOFILL_NO_EXCLUDE_DESCENDANTS;
        int count = (excludedHere && node.getAutofillType() == View.AUTOFILL_TYPE_TEXT) ? 1 : 0;
        boolean excludeChildren = ancestorExcluded
            || importance == View.IMPORTANT_FOR_AUTOFILL_NO_EXCLUDE_DESCENDANTS
            || importance == View.IMPORTANT_FOR_AUTOFILL_YES_EXCLUDE_DESCENDANTS;
        for (int i = 0; i < node.getChildCount(); i++) {
            count += countExcluded(node.getChildAt(i), excludeChildren);
        }
        return count;
    }

    @Override public void onFillRequest(FillRequest request, CancellationSignal cancel, FillCallback callback) {
        long received = SystemClock.uptimeMillis();
        int flags = request.getFlags();
        boolean manual = (flags & FillRequest.FLAG_MANUAL_REQUEST) != 0;
        List<FillContext> contexts = request.getFillContexts();
        if (contexts.isEmpty()) { callback.onSuccess(null); return; }
        FillContext ctx = contexts.get(contexts.size() - 1);
        AssistStructure tree = ctx.getStructure();
        if (!TARGET.equals(tree.getActivityComponent())) { callback.onSuccess(null); return; }
        ProbeIme.Focus sample = ProbeIme.active;
        long acquired = tree.getAcquisitionStartTime(), complete = tree.getAcquisitionEndTime();
        android.util.Log.i("MenuFill", "event=callback id=" + request.getId() + " manual=" + manual
            + " flags=" + flags + " ms=" + received + " start=" + acquired + " end=" + complete);
        if (!manual || sample == null || sample != MenuIme.used || !sample.current()
                || !MenuIme.optedIn(sample.owner, sample.owner.getCurrentInputEditorInfo())) {
            callback.onSuccess(null);
            return;
        }
        Observation o = new Observation();
        o.before = true;
        int excludedText = 0;
        try {
            excludedText = countExcludedTextNodes(tree);
            if (ctx.getRequestId() != request.getId() || acquired <= MenuIme.issuedAt) {
                o.reason = "STALE_FILL_CONTEXT";
            } else {
                o.reason = o.evaluate(ctx, sample);
                if (o.editors.isEmpty() && excludedText > 0 && "EDITOR_COUNT".equals(o.reason)) {
                    o.reason = "TARGET_OPTED_OUT";
                }
            }
        } catch (ReflectiveOperationException | RuntimeException failed) {
            o.reason = "UNAVAILABLE";
        }
        final int finalExcluded = excludedText;
        final boolean uiShown = sample.owner.isInputViewShown();
        main.postDelayed(() -> {
            o.after = sample.current();
            if (cancel.isCanceled() || !o.after) o.reason = "CONNECTION_CHANGED";
            try {
                JSONObject row = o.receipt().put("ime_generation", sample.generation)
                    .put("request_id", request.getId()).put("issued_at", MenuIme.issuedAt)
                    .put("manual", manual).put("settled_at", SystemClock.uptimeMillis())
                    .put("excluded_text_nodes", finalExcluded)
                    .put("mode", MenuIme.lastMode).put("ui_shown", uiShown);
                android.util.Log.i("MenuFill", "result=" + row);
            } catch (JSONException failed) { throw new IllegalStateException(failed); }
            if (!cancel.isCanceled()) callback.onSuccess(null);
        }, 100);
    }
    @Override public void onSaveRequest(SaveRequest request, SaveCallback callback) {
        callback.onFailure("synthetic fixture has no save contract");
    }
